// Shop & Inventory Systems
const Shop = {
    renderShop() {
        const list = document.getElementById('shop-items');
        list.innerHTML = '';
        
        // Seeds
        Object.values(PLANTS).forEach(plant => {
            const div = document.createElement('div');
            div.className = 'list-item';
            div.innerHTML = `
                <div class="item-icon">${plant.emoji}</div>
                <div class="item-info">
                    <h4>${plant.name} Seed</h4>
                    <p>Costs: ${plant.seedCost} 💰</p>
                </div>
                <button class="btn-action" onclick="Shop.buySeed('${plant.id}', ${plant.seedCost})">Buy</button>
            `;
            list.appendChild(div);
        });

        // Upgrades
        const upgList = document.getElementById('upgrade-items');
        upgList.innerHTML = '';
        const upgrades = [
            { target: 8, cost: 300, name: '8 Plots' },
            { target: 10, cost: 800, name: '10 Plots' }
        ];

        upgrades.forEach(upg => {
            if (Player.data.maxPlots < upg.target && (Player.data.maxPlots === upg.target - 2)) {
                const div = document.createElement('div');
                div.className = 'list-item';
                div.innerHTML = `
                    <div class="item-icon">🏡</div>
                    <div class="item-info">
                        <h4>Expand Farm</h4>
                        <p>Costs: ${upg.cost} 💰</p>
                    </div>
                    <button class="btn-action" onclick="Shop.buyUpgrade(${upg.target}, ${upg.cost})">Upgrade</button>
                `;
                upgList.appendChild(div);
            }
        });
        
        if(upgList.innerHTML === '') upgList.innerHTML = '<p style="text-align:center;color:#666;">Max farm reached!</p>';
    },

    buySeed(plantId, cost) {
        if (Player.data.coins >= cost) {
            Player.addCoins(-cost);
            Player.data.inventory[plantId] = (Player.data.inventory[plantId] || 0) + 1;
            Player.save();
            AudioSys.playCoin();
        } else {
            alert("Not enough coins! Harvest crops to earn more.");
        }
    },

    buyUpgrade(targetPlots, cost) {
        if (Player.data.coins >= cost) {
            Player.addCoins(-cost);
            Farm.upgradePlots(targetPlots);
            AudioSys.playLevelUp();
            this.renderShop(); // Refresh view
        } else {
            alert("Not enough coins!");
        }
    },

    renderInventory(isPlantingMode = false) {
        const list = document.getElementById('inventory-items');
        const hint = document.getElementById('inventory-hint');
        list.innerHTML = '';
        
        hint.style.display = isPlantingMode ? 'block' : 'none';

        let hasItems = false;
        Object.keys(Player.data.inventory).forEach(plantId => {
            const count = Player.data.inventory[plantId];
            if (count > 0) {
                hasItems = true;
                const plant = PLANTS[plantId];
                const div = document.createElement('div');
                div.className = 'list-item';
                div.innerHTML = `
                    <div class="item-icon">${plant.emoji}</div>
                    <div class="item-info">
                        <h4>${plant.name} Seeds</h4>
                        <p>Owned: ${count}</p>
                    </div>
                    ${isPlantingMode ? `<button class="btn-action" onclick="Farm.plant('${plantId}')">Plant</button>` : ''}
                `;
                list.appendChild(div);
            }
        });

        if (!hasItems) {
            list.innerHTML = '<p style="text-align:center; margin-top:20px;">Your bag is empty. Buy seeds in the shop!</p>';
        }
    }
};
