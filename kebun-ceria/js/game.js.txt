// Main Game Controller
const Game = {
    init() {
        // Wire up static UI buttons
        document.getElementById('btn-start').addEventListener('click', () => {
            AudioSys.playClick();
            document.getElementById('splash-screen').classList.add('hidden');
            document.getElementById('game-ui').classList.remove('hidden');
            DailyReward.check();
        });

        document.querySelectorAll('.nav-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                AudioSys.playClick();
                const target = btn.closest('button').dataset.target;
                if(target === 'shop-modal') Shop.renderShop();
                if(target === 'inventory-modal') Shop.renderInventory(false);
                if(target === 'quests-modal') Quests.render();
                this.openModal(target);
            });
        });

        document.querySelectorAll('.btn-close').forEach(btn => {
            btn.addEventListener('click', (e) => {
                AudioSys.playClick();
                e.target.closest('.modal').classList.add('hidden');
            });
        });

        document.getElementById('btn-claim-daily').addEventListener('click', () => DailyReward.claim());
        document.getElementById('btn-level-ok').addEventListener('click', () => {
            AudioSys.playClick();
            this.closeModal('level-up-modal');
        });

        // Initialize Systems
        Player.load();
        Farm.load();
        Quests.load();
    },

    openModal(id) {
        document.getElementById(id).classList.remove('hidden');
    },

    closeModal(id) {
        document.getElementById(id).classList.add('hidden');
    },

    openInventoryForPlanting() {
        Shop.renderInventory(true);
        this.openModal('inventory-modal');
    }
};

// Boot Game
window.onload = () => {
    Game.init();
};
