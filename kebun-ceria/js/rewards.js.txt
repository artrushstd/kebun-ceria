// Daily Reward System
const DailyReward = {
    check() {
        const lastRewardTime = Storage.load('lastDailyReward', 0);
        const now = Date.now();
        const oneDay = 24 * 60 * 60 * 1000;

        if (now - lastRewardTime > oneDay) {
            Game.openModal('daily-reward-modal');
        }
    },

    claim() {
        Storage.save('lastDailyReward', Date.now());
        Player.addCoins(50);
        AudioSys.playCoin();
        Game.closeModal('daily-reward-modal');
    }
};
