// Procedural Audio System (No external files needed)
const AudioSys = {
    ctx: null,
    enabled: true,

    init() {
        try {
            const AudioContext = window.AudioContext || window.webkitAudioContext;
            this.ctx = new AudioContext();
        } catch (e) {
            this.enabled = false;
        }
    },

    playTone(freq, type, duration, vol=0.1) {
        if (!this.enabled || !this.ctx) return;
        if (this.ctx.state === 'suspended') this.ctx.resume();

        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        
        osc.type = type;
        osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
        
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        
        osc.start();
        gain.gain.setValueAtTime(vol, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.00001, this.ctx.currentTime + duration);
        osc.stop(this.ctx.currentTime + duration);
    },

    playClick() { this.playTone(600, 'sine', 0.1, 0.1); },
    playPlant() { this.playTone(300, 'triangle', 0.1, 0.2); setTimeout(()=>this.playTone(400, 'triangle', 0.1, 0.2), 100); },
    playHarvest() { this.playTone(800, 'square', 0.1, 0.1); setTimeout(()=>this.playTone(1200, 'square', 0.1, 0.1), 100); },
    playCoin() { this.playTone(1046.50, 'sine', 0.1, 0.1); setTimeout(()=>this.playTone(1318.51, 'sine', 0.2, 0.1), 100); },
    playLevelUp() { 
        [523.25, 659.25, 783.99, 1046.50].forEach((f, i) => {
            setTimeout(() => this.playTone(f, 'square', 0.2, 0.1), i * 150);
        });
    }
};

document.addEventListener('click', () => {
    if(!AudioSys.ctx) AudioSys.init();
}, { once: true });
